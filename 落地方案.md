# 个人技术站点内容工程化落地方案

## 1. 背景与目标

当前站点已经具备较强的内容资产和专业技术表达能力，结构上也比较成熟，主要文件包括：

- [README.md](./README.md)
- [index.html](./index.html)
- [blog.html](./blog.html)
- [deep.html](./deep.html)
- [gen_articles.py](./gen_articles.py)
- [gen_deep.py](./gen_deep.py)
- [verify_titles.py](./verify_titles.py)

但当前站点仍存在几个关键问题：

1. 文章元数据不统一
2. 内容索引和分类依赖人工维护
3. 搜索、标签、归档和 RSS 能力缺失
4. 随着内容增长，维护成本增高
5. 自动化校验还不完整

因此，后续最重要的工作不是“重做页面”，而是把站点升级成“内容工程化”的知识库：

- 统一文章结构
- 自动生成索引
- 让标签、归档、搜索可持续扩展
- 用脚本持续校验站点稳定性

最终目标是：

- 让站点从“内容展示页”升级成“个人技术品牌 + 知识库”
- 让内容可维护、可发现、可扩展
- 降低长期运营成本

## 2. 总体实施思路

建议按如下顺序推进：

1. 统一文章元数据
2. 改造生成脚本，支持 front matter 和索引生成
3. 增加标签页、归档页和搜索能力

5. 加入 CI 自动校验

这条路线最稳妥，也最符合当前纯静态站点的特性。

## 4. [gen_articles.py](./gen_articles.py) 改造方案

这是最关键的生成脚本，建议增加“front matter 解析 → 元数据构造 → 页面渲染 → 索引输出”四层流程。

### 4.1 增加 front matter 解析

新增函数：

```python
# -*- coding: utf-8 -*-
import os
import re
import html
import json
import markdown
from pathlib import Path
from pygments.formatters import HtmlFormatter

try:
    import yaml
except ImportError:
    yaml = None

ROOT = Path(__file__).resolve().parent


def parse_front_matter(text):
    if not text.startswith("---\n"):
        return {}, text

    end_marker = "\n---\n"
    end = text.find(end_marker, 4)
    if end == -1:
        return {}, text

    front = text[4:end]
    body = text[end + len(end_marker):]
    if yaml is None:
        return {}, body

    try:
        data = yaml.safe_load(front) or {}
        if not isinstance(data, dict):
            return {}, body
        return data, body
    except Exception:
        return {}, body
```

## 5. 完整的 generate_index.py 代码

新建文件：

- [generate_index.py](./generate_index.py)

代码如下：

```python
# -*- coding: utf-8 -*-
import os
import json
import re
from pathlib import Path
from collections import defaultdict

ROOT = Path(__file__).resolve().parent
DATA_DIR = ROOT / "data"
DATA_DIR.mkdir(exist_ok=True)


def parse_front_matter(text):
    if not text.startswith("---\n"):
        return {}, text

    end = text.find("\n---\n", 4)
    if end == -1:
        return {}, text

    front = text[4:end]
    body = text[end + len("\n---\n"):]
    try:
        import yaml
        data = yaml.safe_load(front) or {}
        if not isinstance(data, dict):
            return {}, body
        return data, body
    except Exception:
        return {}, body


def get_title_from_md(body):
    m = re.search(r"^#\s+(.+?)\s*$", body, re.M)
    if m:
        return m.group(1).strip()
    return None


def collect_md_files():
    files = []
    for base in [ROOT / "blog", ROOT / "deep"]:
        if not base.exists():
            continue
        for p in base.rglob("*.md"):
            if p.name.startswith("."):
                continue
            files.append(p)
    return sorted(files)


def infer_series_from_path(md_path):
    rel = str(md_path.relative_to(ROOT))
    if rel.startswith("blog/"):
        return "博客"
    if rel.startswith("deep/"):
        parts = rel.split(os.sep)
        if len(parts) >= 2:
            return parts[1]
    return "其它"


def build_record(md_path):
    text = md_path.read_text(encoding="utf-8")
    meta, body = parse_front_matter(text)

    title = meta.get("title") or get_title_from_md(body) or md_path.stem
    summary = meta.get("summary") or ""
    date = meta.get("date") or ""
    tags = meta.get("tags") or []
    category = meta.get("category") or ("blog" if "blog/" in str(md_path.relative_to(ROOT)) else "deep")
    series = meta.get("series") or infer_series_from_path(md_path)
    slug = meta.get("slug") or md_path.stem
    reading_time = meta.get("reading_time") or 6

    rel = md_path.relative_to(ROOT).as_posix()
    url = "/" + rel.replace(".md", ".html")

    return {
        "title": title,
        "summary": summary,
        "date": date,
        "tags": tags,
        "category": category,
        "series": series,
        "slug": slug,
        "reading_time": reading_time,
        "url": url,
        "source": rel,
    }


def build_articles_json():
    records = [build_record(p) for p in collect_md_files()]
    records = sorted(records, key=lambda x: x.get("date") or "", reverse=True)
    (DATA_DIR / "articles.json").write_text(
        json.dumps(records, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )
    return records


def build_tags_json(records):
    tag_map = defaultdict(list)
    for r in records:
        for tag in r.get("tags", []):
            tag_map[tag].append(r)

    tags = [{"name": name, "count": len(items)} for name, items in sorted(tag_map.items())]
    (DATA_DIR / "tags.json").write_text(
        json.dumps(tags, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )
    return tags


def build_archives_json(records):
    year_map = defaultdict(list)
    for r in records:
        date = r.get("date")
        if not date:
            continue
        year = str(date)[:4]
        year_map[year].append(r)

    archives = [{"year": year, "items": items} for year, items in sorted(year_map.items(), reverse=True)]
    (DATA_DIR / "archives.json").write_text(
        json.dumps(archives, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )
    return archives


def build_search_index(records):
    index = []
    for r in records:
        entry = {
            "title": r["title"],
            "summary": r["summary"],
            "tags": r.get("tags", []),
            "url": r["url"],
            "series": r["series"],
            "category": r["category"],
            "date": r["date"],
            "keywords": [r["title"], r["series"], *r.get("tags", [])]
        }
        index.append(entry)

    (DATA_DIR / "search_index.json").write_text(
        json.dumps(index, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )
    return index


def build_rss(records):
    items = []
    for r in records[:20]:
        item = f"""
        <item>
          <title>{r['title']}</title>
          <link>https://liyonggui.github.io{r['url']}</link>
          <description>{r['summary']}</description>
          <pubDate>{r['date']}</pubDate>
        </item>
        """.strip()
        items.append(item)

    rss = """<?xml version=\"1.0\" encoding=\"UTF-8\"?>
<rss version=\"2.0\">
  <channel>
    <title>小番茄程序猿 · 技术站点</title>
    <link>https://liyonggui.github.io/website/</link>
    <description>技术博客与深度系列文章</description>
    """ + "\n".join(items) + """
  </channel>
</rss>
    """
    (ROOT / "rss.xml").write_text(rss, encoding="utf-8")


def main():
    records = build_articles_json()
    build_tags_json(records)
    build_archives_json(records)
    build_search_index(records)
    build_rss(records)
    print(f"Generated {len(records)} records")


if __name__ == "__main__":
    main()
```

这个脚本会让站点具备：

- 文章列表索引
- 标签数据
- 归档数据
- 搜索索引
- RSS 订阅

## 6. 完整的 verify_titles.py 升级版

原来的脚本主要校验标题和正文 H1；建议升级成更全面的站点内容校验脚本。

代码如下：

```python
# -*- coding: utf-8 -*-
import glob
import io
import re
import sys
from pathlib import Path

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
ROOT = Path(__file__).resolve().parent


def parse_front_matter(text):
    if not text.startswith("---\n"):
        return {}, text
    end = text.find("\n---\n", 4)
    if end == -1:
        return {}, text
    front = text[4:end]
    body = text[end + len("\n---\n"):]
    meta = {}
    for line in front.splitlines():
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        if ":" in line:
            key, val = line.split(":", 1)
            meta[key.strip()] = val.strip().strip('"').strip("'")
    return meta, body


def check_md_quality():
    errors = []
    for path in list(ROOT.glob("blog/*.md")) + list(ROOT.glob("deep/*/*.md")):
        text = path.read_text(encoding="utf-8")
        meta, body = parse_front_matter(text)

        if not meta.get("title"):
            errors.append(f"{path}: 缺少 title")
        if not meta.get("summary"):
            errors.append(f"{path}: 缺少 summary")
        if not meta.get("date"):
            errors.append(f"{path}: 缺少 date")
        if not meta.get("tags"):
            errors.append(f"{path}: 缺少 tags")

        if not re.search(r"^#\s+.+", body, re.M):
            errors.append(f"{path}: 正文未包含 H1 标题")

    return errors


def check_html_quality():
    errors = []
    for path in glob.glob("blog/*.html") + glob.glob("deep/*/*.html"):
        html = open(path, encoding="utf-8").read()
        head = re.findall(r'<div class="article-head">.*?<h1>(.*?)</h1>', html, re.S)
        m = re.search(r'<article class="prose">(.*?)</article>', html, re.S)
        body_h1 = m.group(1).count("<h1") if m else -1
        if len(head) != 1 or body_h1 != 0:
            errors.append(f"{path}: 页头 h1 数或正文 h1 数异常")
    return errors


def check_missing_html():
    errors = []
    for md_path in list(ROOT.glob("blog/*.md")) + list(ROOT.glob("deep/*/*.md")):
        html_path = md_path.with_suffix(".html")
        if not html_path.exists():
            errors.append(f"{md_path}: 对应 HTML 未生成")
    return errors


def main():
    errors = []
    errors.extend(check_md_quality())
    errors.extend(check_html_quality())
    errors.extend(check_missing_html())

    if errors:
        print("站点校验失败：")
        for e in errors:
            print(" -", e)
        raise SystemExit(1)

    print("站点校验通过")


if __name__ == "__main__":
    main()
```

这一版校验非常关键，因为它不仅校验页面结构，还校验文章的元数据完整性；对于长期维护十分重要。

## 7. GitHub Actions 自动校验方案

新增文件：

- [.github/workflows/site-check.yml](./.github/workflows/site-check.yml)

内容如下：

```yaml
name: Site Check

on:
  push:
    branches: [ main ]
  pull_request:

jobs:
  validate:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install PyYAML markdown pygments

      - name: Generate pages
        run: |
          python gen_articles.py
          python gen_deep.py
          python generate_index.py

      - name: Validate site
        run: |
          python verify_titles.py
```

这个方案可以让你在每次提交时自动校验：

- 页面是否生成成功
- 标题是否异常
- 文章元数据是否完整
- HTML 是否对应存在

这对长期维护非常重要。
